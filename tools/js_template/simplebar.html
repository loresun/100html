📝 使用方法

<!-- 引入 CSS -->
<link rel="stylesheet" href="simplebar.css"/>

<!-- HTML (自动初始化) -->
<div data-simplebar style="max-height: 300px">...</div>

<!-- JS 手动初始化 -->
new SimpleBar(document.getElementById('my-element'), {
  autoHide: false,
  scrollbarMinSize: 20
});

<!-- 自定义 CSS -->
.simplebar-scrollbar::before {
  background-color: white;
}
