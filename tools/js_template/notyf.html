📝 代码示例

// 创建 Notyf 实例
const notyf = new Notyf({
  duration: 3000,
  position: { x: 'right', y: 'bottom' },
  dismissible: true,
  ripple: true
});

// 显示成功通知
notyf.success('操作成功完成！');

// 显示错误通知
notyf.error('发生了一个错误！');

// 自定义通知
notyf.open({
  type: 'warning',
  message: '这是一条警告信息',
  duration: 5000,
  background: '#f59e0b'
});
