📝 代码示例

import { set, get, del, clear, keys, entries } from 'idb-keyval';

// 存储
await set('user', { name: '张三', age: 25 });

// 读取
const user = await get('user');

// 删除
await del('user');

// 获取所有键
const allKeys = await keys();

// 获取所有键值对
const allEntries = await entries();

// 清空
await clear();
