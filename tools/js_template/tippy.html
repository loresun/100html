📝 代码示例

// 基础用法
tippy('#element', {
  content: '这是提示内容',
  placement: 'top',
  theme: 'light'
});

// HTML 内容
tippy('#element', {
  content: '<strong>HTML</strong> 内容',
  allowHTML: true,
  interactive: true
});

// 动画效果
tippy('#element', {
  animation: 'scale',
  duration: [300, 200]
});
