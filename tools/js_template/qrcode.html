📝 代码示例

// 基础用法
new QRCode(document.getElementById("qrcode"), "https://example.com");

// 高级配置
new QRCode(document.getElementById("qrcode"), {
  text: "https://example.com",
  width: 256,
  height: 256,
  colorDark: "#000000",
  colorLight: "#ffffff",
  correctLevel: QRCode.CorrectLevel.H
});

// 更新内容
qrcode.clear();
qrcode.makeCode("新内容");
