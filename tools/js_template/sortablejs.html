📝 代码示例

// 基础用法
new Sortable(document.getElementById('list'), {
  animation: 150,
  ghostClass: 'sortable-ghost'
});

// 共享列表
new Sortable(todoList, {
  group: 'shared',
  animation: 150
});

// 带手柄
new Sortable(list, {
  handle: '.handle',
  animation: 150
});

// 事件监听
new Sortable(list, {
  onEnd: function(evt) {
    console.log('从', evt.oldIndex, '移动到', evt.newIndex);
  }
});
