📝 代码示例

// 基础动画
gsap.to(".box", { x: 200, duration: 1, ease: "power2.out" });

// 时间轴
const tl = gsap.timeline();
tl.to(".box1", { x: 100 })
  .to(".box2", { x: 100 }, "-=0.5")  // 提前0.5秒开始
  .to(".box3", { x: 100 }, "+=0.2"); // 延迟0.2秒

// 交错动画
gsap.to(".items", {
  y: 50,
  opacity: 1,
  stagger: { amount: 1, from: "center" }
});

// ScrollTrigger (需要插件)
gsap.to(".section", {
  scrollTrigger: ".section",
  x: 400
});
