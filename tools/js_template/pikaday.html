🌐 国际化与格式 (中文 & 农历风格)

📝 代码示例

// 基础用法
new Pikaday({ field: document.getElementById('datepicker') });

// 中文配置
new Pikaday({
  field: document.getElementById('cn-picker'),
  i18n: {
    previousMonth: '上个月',
    nextMonth: '下个月',
    months: ['一月','二月','三月','四月','五月','六月','七月','八月','九月','十月','十一月','十二月'],
    weekdays: ['周日','周一','周二','周三','周四','周五','周六'],
    weekdaysShort: ['日','一','二','三','四','五','六']
  },
  format: 'YYYY-MM-DD'
});
