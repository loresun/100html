📝 代码示例

// 初始化 Clipboard.js
const clipboard = new ClipboardJS('.copy-btn');

// 复制成功回调
clipboard.on('success', function(e) {
  console.log('已复制:', e.text);
  e.clearSelection(); // 清除选中状态
});

// 复制失败回调
clipboard.on('error', function(e) {
  console.error('复制失败');
});

// 使用 data-clipboard-text 属性
// <button data-clipboard-text="要复制的文本">复制</button>

// 使用 data-clipboard-target 属性
// <button data-clipboard-target="#input-id">复制</button>
