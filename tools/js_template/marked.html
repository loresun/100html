📝 使用方法

    

      
// 基础用法
const html = marked.parse('# Hello World');

// 配置选项
marked.setOptions({
  gfm: true,
  breaks: false,
  headerIds: true
});

// 自定义渲染器
const renderer = new marked.Renderer();
renderer.link = (href, title, text) => {
  return `<a href="${href}" target="_blank">${text}</a>`;
};
marked.use({ renderer });
