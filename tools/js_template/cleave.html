📝 代码示例

// 信用卡格式
new Cleave('#card-number', {
  creditCard: true,
  onCreditCardTypeChanged: function(type) {
    console.log('卡类型:', type);
  }
});

// 电话号码
new Cleave('#phone', {
  phone: true,
  phoneRegionCode: 'CN'
});

// 自定义格式
new Cleave('#custom', {
  blocks: [4, 4, 4, 4],
  delimiter: '-',
  numericOnly: true
});
