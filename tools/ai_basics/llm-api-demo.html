1 什么是 API？

API = 应用程序编程接口，简单说就是：

一个"窗口"，让你的程序可以"调用"远程服务的能力。

类比：餐厅

你 = 你的程序
服务员 = API
厨房 = 大模型服务器
菜单 = API 文档

调用流程

你的程序
API 服务器
AI 模型

2 API 调用的基本结构

1. 配置 
{
  endpoint: 'https://...',
  apiKey: 'sk-xxx',
  model: 'deepseek-v4-flash'
}
2. 请求体 
{
  model: 'deepseek-v4-flash',
  messages: [
    {role: 'user', content: '你好'}
  ]
}
3. 响应 
{
  choices: [{
    message: {
      content: '你好！我是...'
    }
  }]
}
提示：每次调用 API 都需要发送 messages 数组，里面包含对话历史。模型会根据历史来理解上下文。

}

3.3 并行调用多个请求 
// 并行调用：同时发送多个请求，提高效率
async function parallelDemo() {
    const llm = createLLM('sk-xxx');
    
    // 使用 Promise.all 并行执行
    const [result1, result2, result3] = await Promise.all([
        llm.chat('用一句话介绍北京'),
        llm.chat('用一句话介绍上海'),
        llm.chat('用一句话介绍深圳')
    ]);
    
    console.log('北京:', result1.content);
    console.log('上海:', result2.content);
    console.log('深圳:', result3.content);
}

// 错误处理：即使一个失败，其他继续
async function safeParallel() {
    const results = await Promise.allSettled([
        llm.chat('问题1'),
        llm.chat('问题2'),
        llm.chat('问题3')
    ]);
    
    results.forEach((r, i) => {
        if (r.status === 'fulfilled') {
            console.log(`问题${i+1}:`, r.value.content);
        } else {
            console.log(`问题${i+1} 失败:`, r.reason.message);
        }
    });
}
3.4 进阶：自定义请求参数 
// 调整 temperature（创造性）
await llm.chat('写一首诗', { temperature: 0.9 });  // 更随机

// 限制回复长度
await llm.chat('详细解释', { maxTokens: 100 });   // 简短回复

// JSON 模式输出
await llm.chat('返回一个JSON对象', { 
    temperature: 0.3,
    model: 'deepseek-v4-flash'  // 指定模型
});

// 切换不同模型
const openai = createLLM('sk-xxx');
openai.setEndpoint('https://api.openai.com/v1/chat/completions');
openai.setModel('gpt-4');
await openai.chat('你好');
